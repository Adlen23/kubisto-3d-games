   393→function StoryCreatorSection() {
   394→  const { t, isAr } = useLanguage()
   395→  const pc = t.parentCorner.storyCreator
   396→  const [childName, setChildName] = useState('')
   397→  const [childAge, setChildAge] = useState('')
   398→  const [hobbies, setHobbies] = useState('')
   399→  const [storyType, setStoryType] = useState<'adventure' | 'bedtime' | 'educational'>('adventure')
   400→  const [story, setStory] = useState('')
   401→  const [generating, setGenerating] = useState(false)
   402→  const [error, setError] = useState('')
   403→  const [storyHistory, setStoryHistory] = useState<Array<{ title: string; date: string }>>([])
   404→
   405→  const storyTypes = [
   406→    { id: 'adventure' as const, emoji: '🗺️', color: 'from-orange-400 to-amber-500', label: pc.adventureStory || (isAr ? 'قصة مغامرة' : 'Adventure Story') },
   407→    { id: 'bedtime' as const, emoji: '🌙', color: 'from-indigo-400 to-purple-600', label: pc.bedtimeStory || (isAr ? 'قصة قبل النوم' : 'Bedtime Story') },
   408→    { id: 'educational' as const, emoji: '🎓', color: 'from-emerald-400 to-teal-600', label: pc.educationalStory || (isAr ? 'قصة تعليمية' : 'Educational Story') },
   409→  ]
   410→
   411→  const handleGenerate = async () => {
   412→    if (!childName || !childAge || !hobbies) return
   413→    setGenerating(true)
   414→    setError('')
   415→    setStory('')
   416→
   417→    try {
   418→      const res = await fetch('/api/generate-story', {
   419→        method: 'POST',
   420→        headers: { 'Content-Type': 'application/json' },
   421→        body: JSON.stringify({
   422→          childName,
   423→          childAge,
   424→          hobbies,
   425→          storyType,
   426→          language: isAr ? 'ar' : 'en',
   427→        }),
   428→      })
   429→      const data = await res.json()
   430→      if (data.error) {
   431→        setError(pc.errorGenerate)
   432→      } else {
   433→        setStory(data.story)
   434→        setStoryHistory(prev => [{ title: data.story.split('\n')[0]?.substring(0, 50) || (isAr ? 'قصة جديدة' : 'New Story'), date: new Date().toLocaleDateString(isAr ? 'ar-SA' : 'en-US') }, ...prev.slice(0, 4)])
   435→      }
   436→    } catch {
   437→      setError(pc.errorGenerate)
